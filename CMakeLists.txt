cmake_minimum_required(VERSION 3.16)
project(kakao-linux-bridge VERSION 0.4.0 LANGUAGES C)
include(GNUInstallDirs)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS
    OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0>=2.74)
pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4>=4.10)
add_library(bridge-config STATIC src/config.c)
target_link_libraries(bridge-config PUBLIC PkgConfig::GIO)
add_executable(kakao-linux-bridge src/bridge.c src/attention.c src/wine-tuning.c src/prefix-integration.c)
target_link_libraries(kakao-linux-bridge PRIVATE bridge-config PkgConfig::XCB)
add_executable(kakao-linux-bridge-settings src/settings.c src/image-clipboard.c)
target_link_libraries(kakao-linux-bridge-settings PRIVATE bridge-config PkgConfig::GTK)
foreach(target bridge-config kakao-linux-bridge kakao-linux-bridge-settings)
  set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
install(TARGETS kakao-linux-bridge kakao-linux-bridge-settings RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/io.github.KakaoLinuxBridge.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/io.github.KakaoLinuxBridge.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/kakao-linux-bridge)
set(CPACK_GENERATOR DEB)
set(CPACK_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS "${CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS}")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native Ubuntu launcher and settings for official KakaoTalk")
set(CPACK_PACKAGE_CONTACT "Kakao Linux Bridge maintainers")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libc6 (>= 2.34), libglib2.0-0t64 (>= 2.74), libgtk-4-1 (>= 4.10), libxcb1")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "wine, fonts-noto-cjk, fontconfig")
set(CPACK_DEBIAN_PACKAGE_SUGGESTS "ibus-hangul, fonts-nanum")
set(CPACK_DEBIAN_PACKAGE_SECTION "net")
set(CPACK_STRIP_FILES ON)
include(CPack)
enable_testing()
add_test(NAME bridge-lifecycle COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/lifecycle.sh $<TARGET_FILE:kakao-linux-bridge>)
set_tests_properties(bridge-lifecycle PROPERTIES TIMEOUT 30)
find_program(XVFB_EXECUTABLE Xvfb)
if(XVFB_EXECUTABLE)
  add_executable(attention-fixture tests/attention-fixture.c)
  target_link_libraries(attention-fixture PRIVATE PkgConfig::GIO PkgConfig::XCB)
  set_target_properties(attention-fixture PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_compile_options(attention-fixture PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME bridge-attention COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/attention.sh
    $<TARGET_FILE:kakao-linux-bridge> $<TARGET_FILE:attention-fixture> ${XVFB_EXECUTABLE})
  set_tests_properties(bridge-attention PROPERTIES TIMEOUT 30)
endif()
find_program(XCLIP_EXECUTABLE xclip)
if(XVFB_EXECUTABLE AND XCLIP_EXECUTABLE)
  add_executable(clipboard-fixture tests/clipboard-fixture.c src/image-clipboard.c)
  target_include_directories(clipboard-fixture PRIVATE src)
  target_link_libraries(clipboard-fixture PRIVATE PkgConfig::GTK)
  set_target_properties(clipboard-fixture PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_compile_options(clipboard-fixture PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME bridge-image-clipboard COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/clipboard.sh
    $<TARGET_FILE:clipboard-fixture> ${XVFB_EXECUTABLE} ${XCLIP_EXECUTABLE})
  set_tests_properties(bridge-image-clipboard PROPERTIES TIMEOUT 20)
endif()
